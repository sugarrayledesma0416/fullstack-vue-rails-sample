### CENTOS/RHEL ###

# install as root or prefix commands with sudo

# install clam binaries
yum -y install clamav clamd

# yum currently sets permissions on clamav log dir to the wrong user, so you must manually change
# ownership
chown clam:clam /var/log/clamav/

## update virus definitions
freshclam

## set up freshclam to update virus definitions every day
## if you don't do this, clam will start yelling at you when
## you try to scan files that your virus definitions are out of 
## date

## add to crontab (crontab -e)
0 0 * * * /usr/bin/freshclam

## ensure cron daemon service is set to run on startup
chkconfig crond on
