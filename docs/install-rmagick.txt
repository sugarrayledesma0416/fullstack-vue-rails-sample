# 1. install imagemagick package dependencies
yum install freetype
yum install libjpeg
yum install libpng
yum install libwmf
yum install ghostscript

# 2. insteal jpeglib headers
wget http://www.ijg.org/files/jpegsrc.v8.tar.gz
tar xvfz jpegsrc.v8.tar.gz
cd jpeg-8
./configure
make
sudo make install

# 3. download imagemagick package
wget ftp://ftp.fifi.org/pub/ImageMagick/ImageMagick-6.5.8-10.tar.gz

# 4. install imagemagick
tar xvfz ImageMagick-6.5.8-10.tar.gz
cd ImageMagick-6.5.8-10
./configure --with-jpeg=yes
make
sudo make install

# 5. install rmagick
gem install rmagick
